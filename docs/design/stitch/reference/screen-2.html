from causalml.inference.meta import BaseXRegressor
from scipy.stats import logistic

# Load parquet partitions directly from Snowflake Iceberg table

       .filter(pl.col("region") == "EMEA-DACH") \
       .with_columns(treatment = (pl.col("gateway_latency_ms") > 1200).cast(pl.Int32)) \
       .collect()

print(f"Confirmed ATE: {ate:.4f} (95% CI: [{lb:.4f}, {ub:.4f}])")
